}); // FUNGSI BARU: Ganti senarai Bab dalam dropdown berdasarkan Tingkatan window.updateAdminChapterOptions = function() { const formNum = document.getElementById('adm-form').value; const chapterSelect = document.getElementById('adm-chapter-select'); chapterSelect.innerHTML = ''; // Kosongkan subtopik select juga document.getElementById('adm-subtopic-select').innerHTML = ''; const chapters = appSilibusData[formNum]; if (chapters) { for (let chKey in chapters) { let option = document.createElement('option'); option.value = chKey; option.innerText = chapters[chKey].nama; chapterSelect.appendChild(option); } } }; // FUNGSI BARU: Ganti senarai Subtopik dalam dropdown berdasarkan Bab yang dipilih window.updateAdminSubtopicOptions = function() { const formNum = document.getElementById('adm-form').value; const chKey = document.getElementById('adm-chapter-select').value; const subtopicSelect = document.getElementById('adm-subtopic-select'); subtopicSelect.innerHTML = ''; if (chKey && appSilibusData[formNum] && appSilibusData[formNum][chKey] && appSilibusData[formNum][chKey].subtopik) { const subtopics = appSilibusData[formNum][chKey].subtopik; for (let subKey in subtopics) { let option = document.createElement('option'); option.value = subKey; option.innerText = subtopics[subKey].nama; subtopicSelect.appendChild(option); } // Autofill nama Bab dalam input custom sebagai rujukan document.getElementById('adm-chapter-custom').value = appSilibusData[formNum][chKey].nama; } else { document.getElementById('adm-chapter-custom').value = ''; } }; // KEMASKINI FUNGSI: Ambil nilai daripada Select Box atau Custom Input untuk Delete window.deleteContentFromFirebase = function() { const formNum = document.getElementById('adm-form').value; // Ambil dari select jika ada, kalau tak ambil dari custom text const selectedChKey = document.getElementById('adm-chapter-select').value; const customChName = document.getElementById('adm-chapter-custom').value.trim(); const chapterKey = selectedChKey || customChName.replace(/\s+/g, '_'); const chapterRealName = selectedChNameString(formNum, selectedChKey) || customChName; const selectedSubKey = document.getElementById('adm-subtopic-select').value; const customSubName = document.getElementById('adm-subtopic-custom').value.trim(); const subtopicKey = selectedSubKey || customSubName.replace(/\s+/g, '_'); const subtopicRealName = selectedSubNameString(formNum, selectedChKey, selectedSubKey) || customSubName; if (!chapterKey) { alert("Sila pilih atau masukkan nama Bab untuk menentukan data yang mahu dipadam."); return; } let targetPath = `silibus/${formNum}/${chapterKey}`; let mesejPengesahan = `Adakah anda pasti mahu memadam KESELURUHAN Bab "${chapterRealName}" di bawah Tingkatan ${formNum}?`; if (subtopicKey) { targetPath = `silibus/${formNum}/${chapterKey}/subtopik/${subtopicKey}`; mesejPengesahan = `Adakah anda pasti mahu memadam Subtopik "${subtopicRealName}" di bawah Bab "${chapterRealName}"?`; } if (confirm(mesejPengesahan)) { set(ref(db, targetPath), null) .then(() => { alert("Kandungan tersebut telah berjaya dipadamkan daripada Firebase!"); location.reload(); }) .catch((error) => { alert("Gagal memadam kandungan: " + error.message); }); } }; // KEMASKINI FUNGSI: Ambil nilai daripada Select Box atau Custom Input untuk Save window.saveContentToFirebase = function() { const formNum = document.getElementById('adm-form').value; const selectedChKey = document.getElementById('adm-chapter-select').value; const customChName = document.getElementById('adm-chapter-custom').value.trim(); const chapterKey = selectedChKey || customChName.replace(/\s+/g, '_'); const chapterNamaPaparan = selectedChNameString(formNum, selectedChKey) || customChName; const selectedSubKey = document.getElementById('adm-subtopic-select').value; const customSubName = document.getElementById('adm-subtopic-custom').value.trim(); const subtopicKey = selectedSubKey || customSubName.replace(/\s+/g, '_'); const subtopicNamaPaparan = selectedSubNameString(formNum, selectedChKey, selectedSubKey) || customSubName; if(!chapterNamaPaparan || !subtopicNamaPaparan) { alert("Sila pilih atau isi nama Bab dan nama Subtopik."); return; } const targetPath = `silibus/${formNum}/${chapterKey}`; // Gunakan update khusus pada path subtopik supaya data bab sedia ada tidak terpadam keseluruhan jika tambah subtopik baru const subtopicPath = `silibus/${formNum}/${chapterKey}/subtopik/${subtopicKey}`; const babPayload = { nama: chapterNamaPaparan }; const subtopicPayload = { nama: subtopicNamaPaparan, nota: { maksud: document.getElementById('adm-nota-maksud').value, formula: document.getElementById('adm-nota-formula').value, kesilapan: document.getElementById('adm-nota-kesilapan').value, tips: document.getElementById('adm-nota-tips').value, }, contoh: { soalan: document.getElementById('adm-contoh-soalan').value, jalan_kerja: document.getElementById('adm-contoh-jalan').value, }, latihan: { soalan: document.getElementById('adm-soalan-teks').value, a: document.getElementById('adm-opt-a').value, b: document.getElementById('adm-opt-b').value, c: document.getElementById('adm-opt-c').value, d: document.getElementById('adm-opt-d').value, jawapan: document.getElementById('adm-opt-jawapan').value, penerangan: document.getElementById('adm-soalan-penerangan').value, } }; // Simpan nama Bab terlebih dahulu (jika baru), kemudian simpan/kemaskini subtopik update(ref(db, targetPath), babPayload).then(() => { return update(ref(db, subtopicPath), subtopicPayload); }).then(() => { alert("Kandungan berjaya ditambah/dikemaskini ke Firebase database!"); location.reload(); }).catch(err => alert("Gagal menyimpan data: " + err.message)); }; // Helper functions untuk dapatkan nama asal dari state object function selectedChNameString(form, key) { return (appSilibusData[form] && appSilibusData[form][key]) ? appSilibusData[form][key].nama : null; } function selectedSubNameString(form, chKey, subKey) { return (appSilibusData[form] && appSilibusData[form][chKey] && appSilibusData[form][chKey].subtopik) ? appSilibusData[form][chKey].subtopik[subKey].nama : null; }
